Binary Search

#include <stdio.h>

int main()
{
    int a[5] = {10, 20, 30, 40, 50};
    int low = 0, high = 4;
    int key = 30;

    while(low <= high)
    {
        int mid = (low + high) / 2;

        if(a[mid] == key)
        {
            printf("Found");
            break;
        }
        else if(a[mid] < key)
            low = mid + 1;
        else
            high = mid - 1;
    }

    return 0;
}
OUTPUT
Found

Insertion Sort

#include <stdio.h>

int main()
{
    int a[5] = {5, 3, 1, 4, 2};
    int i, j, key;

    for(i = 1; i < 5; i++)
    {
        key = a[i];
        j = i - 1;

        while(j >= 0 && a[j] > key)
        {
            a[j + 1] = a[j];
            j--;
        }

        a[j + 1] = key;
    }

    for(i = 0; i < 5; i++)
        printf("%d ", a[i]);

    return 0;
}

Merge Sort

#include <stdio.h>

void merge(int a[], int l, int m, int r)
{
    int temp[10];
    int i = l, j = m + 1, k = 0;

    while(i <= m && j <= r)
    {
        if(a[i] < a[j])
            temp[k++] = a[i++];
        else
            temp[k++] = a[j++];
    }

    while(i <= m)
        temp[k++] = a[i++];

    while(j <= r)
        temp[k++] = a[j++];

    for(i = l, k = 0; i <= r; i++, k++)
        a[i] = temp[k];
}

void sort(int a[], int l, int r)
{
    int m;

    if(l < r)
    {
        m = (l + r) / 2;

        sort(a, l, m);
        sort(a, m + 1, r);

        merge(a, l, m, r);
    }
}

int main()
{
    int a[5] = {5, 3, 1, 4, 2};
    int i;

    sort(a, 0, 4);

    for(i = 0; i < 5; i++)
        printf("%d ", a[i]);

    return 0;
}
OUTPUT
1 2 3 4 5

Quick Sort

#include <stdio.h>

void quickSort(int a[], int low, int high)
{
    int i = low;
    int j = high;
    int pivot = a[(low + high) / 2];
    int temp;

    while(i <= j)
    {
        while(a[i] < pivot)
            i++;

        while(a[j] > pivot)
            j--;

        if(i <= j)
        {
            temp = a[i];
            a[i] = a[j];
            a[j] = temp;

            i++;
            j--;
        }
    }

    if(low < j)
        quickSort(a, low, j);

    if(i < high)
        quickSort(a, i, high);
}

int main()
{
    int a[5] = {5, 3, 1, 4, 2};
    int i;

    quickSort(a, 0, 4);

    for(i = 0; i < 5; i++)
        printf("%d ", a[i]);

    return 0;
}
OUTPUT
1 2 3 4 5

Selection Sort

#include <stdio.h>

int main()
{
    int a[5] = {5, 3, 1, 4, 2};
    int i, j, min, temp;

    for(i = 0; i < 4; i++)
    {
        min = i;

        for(j = i + 1; j < 5; j++)
        {
            if(a[j] < a[min])
                min = j;
        }

        temp = a[i];
        a[i] = a[min];
        a[min] = temp;
    }

    for(i = 0; i < 5; i++)
        printf("%d ", a[i]);

    return 0;
}
OUTPUT

Factorial using Recursion

#include <stdio.h>

int factorial(int n)
{
    if(n == 0)
        return 1;

    return n * factorial(n - 1);
}

int main()
{
    printf("%d", factorial(5));

    return 0;
}
OUTPUT
120
